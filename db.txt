DROP DATABASE IF EXISTS photo_ate;

CREATE DATABASE photo_ate CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE photo_ate;

CREATE TABLE role (
    role_id INT          AUTO_INCREMENT PRIMARY KEY,
    name    VARCHAR(250) NOT NULL
);

CREATE TABLE type_serv (
    id_type INT          AUTO_INCREMENT PRIMARY KEY,
    name    VARCHAR(250) NOT NULL
);

CREATE TABLE user (
    id_user  INT          AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(250) NOT NULL,
    login    VARCHAR(250) NOT NULL,
    password VARCHAR(250) NOT NULL,
    role_id  INT,
    FOREIGN KEY (role_id) REFERENCES role(role_id)
);

CREATE TABLE service (
    id_service INT          AUTO_INCREMENT PRIMARY KEY,
    naming     VARCHAR(250) NOT NULL,
    descrip    VARCHAR(250),
    price      INT          NOT NULL,
    discount   INT          DEFAULT 0,
    id_type    INT,
    photo      VARCHAR(250),
    FOREIGN KEY (id_type) REFERENCES type_serv(id_type)
);
